import unicodedata


def normalizar(s: str) -> str:
    """minúsculas e sem acento: 'PROGRAMAÇÃO' -> 'programacao' (para busca por nome)."""
    sem_acento = unicodedata.normalize("NFD", s or "")
    return "".join(c for c in sem_acento if unicodedata.category(c) != "Mn").lower()

from __future__ import annotations

import argparse
import json
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent))
import config  # noqa: E402

SCHEMA = config.RAIZ / "db" / "postgres" / "init.sql"
LOGICAS = {"E", "OU"}
TIPOS = {"PRE", "CO", "EQUIV"}
NATUREZAS = {"obrigatoria", "optativa", "atividade_complementar", "outra"}

# Validação

def validar(doc: dict) -> list[str]:
    erros: list[str] = []
    for chave in ("curso", "estrutura", "componentes", "relacoes"):
        if chave not in doc:
            return [f"JSON sem a chave obrigatória '{chave}'"]
    if not doc["curso"].get("sigla") or not doc["curso"].get("nome"):
        erros.append("curso precisa de 'sigla' e 'nome'")
    if not isinstance(doc["estrutura"].get("ano"), int):
        erros.append("estrutura.ano precisa ser inteiro")

    codigos = [c.get("codigo") for c in doc["componentes"]]
    if len(set(codigos)) != len(codigos):
        repetidos = sorted({c for c in codigos if codigos.count(c) > 1})
        erros.append(f"códigos de componente repetidos: {repetidos}")
    for c in doc["componentes"]:
        if not c.get("codigo") or not c.get("nome"):
            erros.append(f"componente sem codigo/nome: {c}")
        if c.get("natureza") not in NATUREZAS:
            erros.append(f"{c.get('codigo')}: natureza inválida '{c.get('natureza')}'")

    conhecidos = set(codigos)
    arestas: dict[str, set[str]] = {}
    vistos = set()
    for r in doc["relacoes"]:
        o, d = r.get("origem"), r.get("destino")
        if o not in conhecidos or d not in conhecidos:
            erros.append(f"relação referencia componente inexistente: {o} -> {d}")
            continue
        if o == d:
            erros.append(f"relação de {o} com ele mesmo")
        if r.get("tipo", "PRE") not in TIPOS or r.get("logica", "E") not in LOGICAS:
            erros.append(f"tipo/logica inválido em {o} -> {d}: {r}")
        chave = (o, d, r.get("tipo", "PRE"))
        if chave in vistos:
            erros.append(f"relação duplicada: {chave}")
        vistos.add(chave)
        if r.get("tipo", "PRE") == "PRE":
            arestas.setdefault(d, set()).add(o)

    # pré-requisitos formam um DAG: ciclo = erro de extração
    estado: dict[str, int] = {}

    def tem_ciclo(no: str) -> bool:
        estado[no] = 1
        for p in arestas.get(no, ()):
            if estado.get(p) == 1 or (estado.get(p) is None and tem_ciclo(p)):
                return True
        estado[no] = 2
        return False

    if any(estado.get(n) is None and tem_ciclo(n) for n in list(arestas)):
        erros.append("ciclo nos pré-requisitos")
    return erros

ESTRUTURA_ID = """(SELECT e.id FROM estrutura_curricular e
                   JOIN curso cu ON cu.id = e.curso_id
                   WHERE cu.sigla = %(sigla)s AND e.ano = %(ano)s)"""


def comandos(doc: dict) -> list[tuple[str, dict]]:
    p = {
        "sigla": doc["curso"]["sigla"],
        "nome": doc["curso"]["nome"],
        "instituicao": doc["curso"].get("instituicao"),
        "ano": doc["estrutura"]["ano"],
        "fonte": doc["estrutura"].get("fonte"),
        "componentes": json.dumps(doc["componentes"], ensure_ascii=False),
        "relacoes": json.dumps(doc["relacoes"], ensure_ascii=False),
    }
    return [
        # 1. curso
        ("""INSERT INTO curso (sigla, nome, instituicao)
            VALUES (%(sigla)s, %(nome)s, %(instituicao)s)
            ON CONFLICT (sigla) DO UPDATE
              SET nome = EXCLUDED.nome, instituicao = EXCLUDED.instituicao""", p),
        # 2. estrutura (curso + ano)
        ("""INSERT INTO estrutura_curricular (curso_id, ano, fonte)
            SELECT id, %(ano)s, %(fonte)s FROM curso WHERE sigla = %(sigla)s
            ON CONFLICT (curso_id, ano) DO UPDATE SET fonte = EXCLUDED.fonte""", p),
        # 3. limpa a versão anterior desta estrutura (as relações saem em cascata)
        (f"DELETE FROM estrutura_componente WHERE estrutura_id = {ESTRUTURA_ID}", p),
        # 4. identidade dos componentes no curso
        ("""INSERT INTO componente (curso_id, codigo)
            SELECT cu.id, x.codigo
            FROM curso cu,
                 jsonb_to_recordset(%(componentes)s::jsonb) AS x(codigo text)
            WHERE cu.sigla = %(sigla)s
            ON CONFLICT (curso_id, codigo) DO NOTHING""", p),
        # 5. atributos dos componentes nesta estrutura
        (f"""INSERT INTO estrutura_componente
               (estrutura_id, componente_id, nome, carga, unidade_carga, natureza, periodo)
            SELECT {ESTRUTURA_ID}, c.id, x.nome, x.carga, x.unidade_carga, x.natureza, x.periodo
            FROM curso cu
            CROSS JOIN jsonb_to_recordset(%(componentes)s::jsonb)
                 AS x(codigo text, nome text, carga int, unidade_carga text,
                      natureza text, periodo int)
            JOIN componente c ON c.curso_id = cu.id AND c.codigo = x.codigo
            WHERE cu.sigla = %(sigla)s""", p),
        # 6. relações (pré-requisitos)
        (f"""INSERT INTO relacao_componente
               (estrutura_id, origem_id, destino_id, tipo, logica)
            SELECT {ESTRUTURA_ID}, o.id, d.id, COALESCE(r.tipo, 'PRE'), COALESCE(r.logica, 'E')
            FROM curso cu
            CROSS JOIN jsonb_to_recordset(%(relacoes)s::jsonb)
                 AS r(origem text, destino text, tipo text, logica text)
            JOIN componente o ON o.curso_id = cu.id AND o.codigo = r.origem
            JOIN componente d ON d.curso_id = cu.id AND d.codigo = r.destino
            WHERE cu.sigla = %(sigla)s""", p),
    ]


RESUMO = """
SELECT cu.sigla, e.ano,
       (SELECT count(*) FROM estrutura_componente ec WHERE ec.estrutura_id = e.id),
       (SELECT count(*) FROM relacao_componente  r  WHERE r.estrutura_id  = e.id)
FROM estrutura_curricular e JOIN curso cu ON cu.id = e.curso_id
ORDER BY cu.sigla, e.ano
"""

SCHEMA = [

    "CREATE CONSTRAINT estrutura_id IF NOT EXISTS FOR (e:Estrutura) REQUIRE e.id IS UNIQUE",

    "CREATE TEXT INDEX componente_nome_txt IF NOT EXISTS FOR (c:Componente) ON (c.nome_norm)",

def montar(doc: dict) -> dict:
    """Função pura: JSON canônico -> parâmetros das queries Cypher."""
    sigla, ano = doc["curso"]["sigla"], doc["estrutura"]["ano"]
    nid = lambda codigo: f"{sigla}:{ano}:{codigo}"
 
    nos = [{
        "id": nid(c["codigo"]), "curso": sigla, "ano": ano, "codigo": c["codigo"],
        "nome": c["nome"], "nome_norm": normalizar(c["nome"]),
        "carga": c.get("carga"), "unidade_carga": c.get("unidade_carga"),
        "natureza": c["natureza"], "periodo": c.get("periodo"),
    } for c in doc["componentes"]]
 
    rels: dict[str, list[dict]] = {}
    for r in doc["relacoes"]:
        tipo = TIPO_REL[r.get("tipo", "PRE")]
        # aresta sai da disciplina que EXIGE e aponta para o pré-requisito
        rels.setdefault(tipo, []).append({
            "de": nid(r["destino"]), "para": nid(r["origem"]), "logica": r.get("logica", "E")})
 
    return {
        "curso": {"sigla": sigla, "nome": doc["curso"]["nome"], "instituicao": doc["curso"].get("instituicao")},
        "estrutura": {"id": f"{sigla}:{ano}", "curso": sigla, "ano": ano, "fonte": doc["estrutura"].get("fonte")},
        "nos": nos,
        "rels": rels,
    }
 
 
def carregar(tx, p: dict) -> None:
    tx.run("MATCH (c:Componente {curso: $sigla, ano: $ano}) DETACH DELETE c",
           sigla=p["estrutura"]["curso"], ano=p["estrutura"]["ano"])
    tx.run("""MERGE (cu:Curso {sigla: $curso.sigla})
              SET cu.nome = $curso.nome, cu.instituicao = $curso.instituicao
              MERGE (e:Estrutura {id: $est.id})
              SET e.curso = $est.curso, e.ano = $est.ano, e.fonte = $est.fonte
              MERGE (cu)-[:TEM_ESTRUTURA]->(e)""", curso=p["curso"], est=p["estrutura"])
    tx.run("""MATCH (e:Estrutura {id: $eid})
              UNWIND $nos AS n
              CREATE (c:Componente) SET c = n
              CREATE (e)-[:INCLUI]->(c)""", eid=p["estrutura"]["id"], nos=p["nos"])
    for tipo, lista in p["rels"].items():            # `tipo` vem da whitelist TIPO_REL
        tx.run(f"""UNWIND $rels AS r
                   MATCH (a:Componente {{id: r.de}}), (b:Componente {{id: r.para}})
                   CREATE (a)-[:{tipo} {{logica: r.logica}}]->(b)""", rels=lista)

from __future__ import annotations

from urllib.parse import quote_plus

def carregar_env(caminho: Path | None = None) -> dict[str, str]:
    """Parser mínimo de .env (CHAVE=valor, # comentários). Ambiente real tem prioridade."""
    valores: dict[str, str] = {}
    arq = caminho or RAIZ / ".env"
    if arq.exists():
        for linha in arq.read_text(encoding="utf-8").splitlines():
            linha = linha.strip()
            if not linha or linha.startswith("#") or "=" not in linha:
                continue
            chave, _, valor = linha.partition("=")
            valores[chave.strip()] = valor.strip().strip('"').strip("'")
    valores.update({k: v for k, v in os.environ.items() if k in valores or k.startswith(("POSTGRES_", "MONGO_", "NEO4J_"))})
    return valores


def postgres() -> dict:
    e = carregar_env()

-- Modelo relacional (3FN) da estrutura curricular.
-- Idempotente (IF NOT EXISTS): roda no primeiro `docker compose up` (pasta de init do
-- container) e também é reaplicado por etl/load_postgres.py.
--
-- Ideias do modelo:
--   * curso 1:N estrutura_curricular (um curso tem várias versões: 2015, 2019...)
--   * componente = identidade da disciplina no curso (curso + código), igual entre anos
--   * estrutura_componente = atributos da disciplina NAQUELA estrutura (nome, carga, natureza...)
--   * relacao_componente = N:N auto-referenciada POR estrutura: os pré-requisitos mudam
--     de um ano para outro (ex.: TAD0029 em 2015 x 2019).
--   * Para outro curso basta inserir outro `curso`: nada mais muda no esquema.

CREATE EXTENSION IF NOT EXISTS pg_trgm;   -- busca por nome parcial com índice

CREATE TABLE IF NOT EXISTS curso (
    id          SERIAL PRIMARY KEY,
    sigla       TEXT NOT NULL UNIQUE,
    nome        TEXT NOT NULL,
    instituicao TEXT
);

CREATE TABLE IF NOT EXISTS estrutura_curricular (
    id       SERIAL PRIMARY KEY,
    curso_id INT  NOT NULL REFERENCES curso (id) ON DELETE CASCADE,
    ano      INT  NOT NULL,
    fonte    TEXT,
    UNIQUE (curso_id, ano)
);

CREATE TABLE IF NOT EXISTS componente (
    id       SERIAL PRIMARY KEY,
    curso_id INT  NOT NULL REFERENCES curso (id) ON DELETE CASCADE,
    codigo   TEXT NOT NULL,
    UNIQUE (curso_id, codigo)
);

CREATE TABLE IF NOT EXISTS estrutura_componente (
    estrutura_id  INT  NOT NULL REFERENCES estrutura_curricular (id) ON DELETE CASCADE,
    componente_id INT  NOT NULL REFERENCES componente (id)           ON DELETE CASCADE,
    nome          TEXT NOT NULL,
    carga         INT,
    unidade_carga TEXT,
    natureza      TEXT NOT NULL
        CHECK (natureza IN ('obrigatoria', 'optativa', 'atividade_complementar', 'outra')),
    periodo       INT,
    PRIMARY KEY (estrutura_id, componente_id)
);

-- origem é pré-requisito de destino. tipo: PRE (pré-requisito), CO (co-requisito),
-- EQUIV (equivalência). logica: E (todas as setas contínuas) ou OU (alternativas).
CREATE TABLE IF NOT EXISTS relacao_componente (
    estrutura_id INT  NOT NULL,
    origem_id    INT  NOT NULL,
    destino_id   INT  NOT NULL,
    tipo         TEXT NOT NULL DEFAULT 'PRE' CHECK (tipo IN ('PRE', 'CO', 'EQUIV')),
    logica       TEXT NOT NULL DEFAULT 'E'   CHECK (logica IN ('E', 'OU')),
    PRIMARY KEY (estrutura_id, origem_id, destino_id, tipo),
    CHECK (origem_id <> destino_id),
    -- as duas pontas precisam pertencer à MESMA estrutura
    FOREIGN KEY (estrutura_id, origem_id)
        REFERENCES estrutura_componente (estrutura_id, componente_id) ON DELETE CASCADE,
    FOREIGN KEY (estrutura_id, destino_id)
        REFERENCES estrutura_componente (estrutura_id, componente_id) ON DELETE CASCADE
);

-- Índices das consultas: descer (pré-requisitos de X) e subir (o que X libera)
CREATE INDEX IF NOT EXISTS idx_rel_destino ON relacao_componente (estrutura_id, destino_id);
CREATE INDEX IF NOT EXISTS idx_rel_origem  ON relacao_componente (estrutura_id, origem_id);
CREATE INDEX IF NOT EXISTS idx_ec_nome_trgm
    ON estrutura_componente USING gin (nome gin_trgm_ops);
CREATE INDEX IF NOT EXISTS idx_ec_natureza ON estrutura_componente (estrutura_id, natureza);
